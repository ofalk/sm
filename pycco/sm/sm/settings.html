            APP_VERSION = (
                subprocess.check_output(
                    ["git", "rev-parse", "--short", "HEAD"], stderr=subprocess.DEVNULL
                )
                .decode("utf-8")
                .strip()
            )
        if APP_MODIFICATION_DATE == "unknown":
            APP_MODIFICATION_DATE = (
                subprocess.check_output(
                    ["git", "log", "-1", "--format=%cd"], stderr=subprocess.DEVNULL
                )
                .decode("utf-8")
                .strip()
            )
